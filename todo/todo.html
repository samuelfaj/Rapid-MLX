T0.4 ✅ — Summarizer

Create scripts/nax_summarize.py:

#!/usr/bin/env python3
"""Compare two bench report directories. Output markdown delta table."""
import json, sys
from pathlib import Path


def load_reports(d):
    return {f.stem: json.loads(f.read_text()) for f in Path(d).rglob("*.json")}


def pct(b, a):
    return 0.0 if b == 0 else ((a - b) / b) * 100.0


def main():
    if len(sys.argv) != 3:
        print("usage: nax_summarize.py <before_dir> <after_dir>", file=sys.stderr); sys.exit(2)
    b, a = load_reports(sys.argv[1]), load_reports(sys.argv[2])
    keys = sorted(set(b) | set(a))
    print("| cell | prefill before | prefill after | Δ% | gen before | gen after | Δ% |")
    print("|------|---------------:|--------------:|---:|-----------:|----------:|---:|")
    for k in keys:
        bb, aa = b.get(k, {}), a.get(k, {})
        pb, pa = bb.get("prefill_tok_s", 0), aa.get("prefill_tok_s", 0)
        gb, ga = bb.get("gen_tok_s", 0), aa.get("gen_tok_s", 0)
        print(f"| {k} | {pb:.2f} | {pa:.2f} | {pct(pb,pa):+.1f}% | {gb:.2f} | {ga:.2f} | {pct(gb,ga):+.1f}% |")


if __name__ == "__main__":
    main()

E9 — nax_runtime_detect low risk

Create vllm_mlx/runtime/metal4.py:

"""Metal 4 / Neural Accelerator runtime detection. Observability only."""
from __future__ import annotations
import functools, logging, os, platform
import mlx.core as mx

logger = logging.getLogger(__name__)


@functools.lru_cache(maxsize=1)
def device_name() -> str:
    try: return mx.device_info().get("device_name", "")
    except Exception: return ""


@functools.lru_cache(maxsize=1)
def macos_version() -> tuple[int, int]:
    parts = platform.mac_ver()[0].split(".")
    try: return int(parts[0]), int(parts[1]) if len(parts) > 1 else 0
    except (ValueError, IndexError): return (0, 0)


@functools.lru_cache(maxsize=1)
def metal4_available() -> bool:
    if os.environ.get("LIGHTNING_DISABLE_METAL4") == "1": return False
    return macos_version() >= (26, 2)


@functools.lru_cache(maxsize=1)
def m5_nax_hint() -> bool:
    return metal4_available() and "M5" in device_name()


def log_capabilities() -> None:
    logger.info("metal4=%s m5_nax=%s device=%s macos=%s",
                metal4_available(), m5_nax_hint(), device_name(), macos_version())

Call log_capabilities() in server startup. Surface in TUI. ≤ 50 LOC total.
